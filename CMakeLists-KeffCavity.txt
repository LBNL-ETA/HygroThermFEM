
# Create include directories that will connect to library itself. This must be created before downloading 
# Windows Calculation Engine library or otherwise connections will not be created
set(CommonIncludeDir "${CMAKE_BINARY_DIR}/KeffCavity-src")
set(KeffCavity_INCLUDE_DIRS "${CommonIncludeDir}/include")

include_directories(${KeffCavity_INCLUDE_DIRS})

# Need to include gases from Windows-CalcEngine
set(CommonIncludeDir "${CMAKE_BINARY_DIR}/Windows-CalcEngine-src/src")
set(WinCalcEngine_INCLUDE_DIRS "${CommonIncludeDir}/Gases/include")

include_directories(${WinCalcEngine_INCLUDE_DIRS})

# Download and create KeffCavity Engine library
configure_file(CMakeLists-KeffCavity.txt.in ${CMAKE_BINARY_DIR}/KeffCavity-download/CMakeLists.txt)
execute_process(COMMAND ${CMAKE_COMMAND} -G "${CMAKE_GENERATOR}" .
  WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/KeffCavity-download)
execute_process(COMMAND ${CMAKE_COMMAND} --build .
  WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/KeffCavity-download)
 
add_subdirectory(${CMAKE_BINARY_DIR}/KeffCavity-src "${CMAKE_CURRENT_BINARY_DIR}/KeffCavity-src")
set(KeffCavity_LIB "${CMAKE_SHARED_LIBRARY_PREFIX}KeffCavity${CMAKE_SHARED_LIBRARY_SUFFIX}")
