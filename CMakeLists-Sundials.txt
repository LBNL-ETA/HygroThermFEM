
set(CommonIncludeDir "${CMAKE_BINARY_DIR}/Sundials-src")
set(Sundials_INCLUDE_DIRS "${CMAKE_BINARY_DIR}/include")

include_directories(${Sundials_INCLUDE_DIRS})

set(BUILD_ARCODE OFF)
set(BUILD_BENCHMARKS OFF)
set(BUILD_CVODE OFF)
set(BUILD_CVODES OFF)
set(BUILD_FORTRAN_MODULE_INTERFACE OFF)
set(BUILD_IDA ON)
set(BUILD_IDAS OFF)
set(BUILD_KINSOL OFF)
set(BUILD_SHARED_LIBS OFF)
set(BUILD_STATIC_LIBS ON)
set(BUILD_TESTING OFF)
set(CMAKE_C_EXTENSIONS OFF)

configure_file(CMakeLists-Sundials.txt.in ${CMAKE_BINARY_DIR}/Sundials-download/CMakeLists.txt)
execute_process(COMMAND ${CMAKE_COMMAND} -G "${CMAKE_GENERATOR}" .
  WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/Sundials-download)
execute_process(COMMAND ${CMAKE_COMMAND} --build .
  WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/Sundials-download)
 
add_subdirectory(${CMAKE_BINARY_DIR}/Sundials-src "${CMAKE_CURRENT_BINARY_DIR}/Sundials-src")
