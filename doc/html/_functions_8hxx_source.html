128  m_Function1(std::move(t)),
129  m_Function2(std::move(s)),
130  m_Operation(op)
131  {
132  m_Operator[Operation::MULT] = [&](double a, double b) { return a * b; };
133  m_Operator[Operation::DIV] = [&](double a, double b) { return a / b; };
134  m_Operator[Operation::ADD] = [&](double a, double b) { return a + b; };
135  m_Operator[Operation::SUB] = [&](double a, double b) { return a - b; };
136  }

140  {
141  return m_Operator.at(m_Operation)(m_Function1.value(node), m_Function2.value(node));
142  }

147  const T m_Function1;
